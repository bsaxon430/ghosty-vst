# Engine-only build (no editor, no JUCE): compiles the synth engine and runs offline sound checks.
# Used on Linux while developing; the plugin itself is built by the main CMakeLists on the Mac.
cmake_minimum_required(VERSION 3.28)
project(seance-headless VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(R ${CMAKE_CURRENT_SOURCE_DIR}/../..)
set(L ${R}/libs)
include(${L}/sst/sst-plugininfra/cmake/git-version-functions.cmake)
version_from_versionfile_or_git(WORKING_DIRECTORY ${R})
add_subdirectory(${L}/sst/sst-cmake sst-cmake)
add_subdirectory(${L}/clap-libs/clap clap EXCLUDE_FROM_ALL)
add_subdirectory(${L}/fmt fmt EXCLUDE_FROM_ALL)
add_library(simde INTERFACE)
target_include_directories(simde INTERFACE ${L}/simde)
add_subdirectory(${L}/sst/sst-cpputils sst-cpputils)
add_subdirectory(${L}/sst/sst-basic-blocks sst-basic-blocks)
add_subdirectory(${L}/sst/sst-voicemanager sst-voicemanager)
add_subdirectory(${L}/sst/sst-filters sst-filters)
set(SST_PLUGININFRA_PROVIDE_TINYXML ON CACHE BOOL "" FORCE)
set(SST_PLUGININFRA_PROVIDE_PATCHBASE ON CACHE BOOL "" FORCE)
set(SST_PLUGININFRA_PROVIDE_MINIZ ON CACHE BOOL "" FORCE)
add_subdirectory(${L}/sst/sst-plugininfra sst-plugininfra)
add_subdirectory(${L}/pffft pffft EXCLUDE_FROM_ALL)
add_library(mts-esp-client STATIC ${L}/MTS-ESP/Client/libMTSClient.cpp)
target_include_directories(mts-esp-client PUBLIC ${L}/MTS-ESP/Client)
set(LIBSAMPLERATE_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory(${L}/libsamplerate libsamplerate EXCLUDE_FROM_ALL)

add_library(engine STATIC
  ${R}/src/dsp/sintable.cpp ${R}/src/dsp/wavetable_build.cpp ${R}/src/dsp/wavetable_io.cpp
  ${R}/src/synth/synth.cpp ${R}/src/synth/voice.cpp ${R}/src/synth/patch.cpp
  ${R}/src/synth/mod_matrix.cpp ${R}/src/synth/macro_usage.cpp)
target_include_directories(engine PUBLIC ${R}/src)
set(SEANCE_BEAST_OBJ "" CACHE FILEPATH "seance-beast.o from src/ghostblock/beast/build-beast.sh")
if (SEANCE_BEAST_OBJ)
  set_source_files_properties(${SEANCE_BEAST_OBJ} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  target_sources(engine PRIVATE ${SEANCE_BEAST_OBJ})
  target_link_libraries(engine PUBLIC dl pthread)
else()
  target_sources(engine PRIVATE ${R}/src/ghostblock/beast/beast-stub.cpp)
endif()
set(SHADE_ROOT ${R})
include(${R}/src/ghostblock/shade/shade.cmake)
target_link_libraries(engine PUBLIC seance-shade)
target_compile_definitions(engine PUBLIC PRODUCT_NAME="Seance")
if (CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
  target_compile_options(engine PUBLIC -msse4.2)
endif()
target_link_libraries(engine PUBLIC clap simde mts-esp-client fmt-header-only sst-basic-blocks sst-voicemanager
  sst-cpputils sst-plugininfra sst-plugininfra::filesystem sst-plugininfra::tinyxml sst-plugininfra::patchbase
  sst-plugininfra::miniz sst-plugininfra::version_information pffft sst-filters samplerate)
if (EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/engine-checks.cpp)
  add_executable(engine-checks engine-checks.cpp)
  target_link_libraries(engine-checks PRIVATE engine)
endif()

# the sound writer (recipes -> .sxsnp), engine only
add_executable(seance-sounds ${R}/src/ghostblock/seance/tools/seance-sounds.cpp)
target_link_libraries(seance-sounds PRIVATE engine)
